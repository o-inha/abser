cmake_minimum_required(VERSION 3.20)
project(Abser VERSION 0.2.1 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(catalog_tests tests/catalog_tests.cpp src/catalog.cpp)
    target_include_directories(catalog_tests PRIVATE src)
    if(MSVC)
        target_compile_options(catalog_tests PRIVATE /W4 /permissive- /utf-8)
    endif()
    if(WIN32)
        target_compile_definitions(catalog_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
        target_link_libraries(catalog_tests PRIVATE shell32 ole32 shlwapi uuid)
    endif()
    add_test(NAME catalog COMMAND catalog_tests)
endif()

if(WIN32)
    enable_language(RC)
    add_executable(Abser WIN32 src/main.cpp src/audio.cpp src/catalog.cpp src/file_drag.cpp src/slim_scrollbars.cpp resources/abser.rc)
    target_include_directories(Abser PRIVATE src third_party resources)
    target_compile_definitions(Abser PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    target_link_libraries(Abser PRIVATE comctl32 shell32 ole32 uuid dwmapi shlwapi comdlg32 winmm avrt uxtheme)
    if(BUILD_TESTING)
        add_executable(abser_audio_tests tests/audio_tests.cpp src/audio.cpp)
        target_include_directories(abser_audio_tests PRIVATE src third_party)
        target_compile_definitions(abser_audio_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
        target_link_libraries(abser_audio_tests PRIVATE winmm ole32 avrt uuid)
        if(MSVC)
            target_compile_options(abser_audio_tests PRIVATE /W4 /permissive- /utf-8)
        endif()
        add_test(NAME audio COMMAND abser_audio_tests)
        set_tests_properties(audio PROPERTIES TIMEOUT 30)
        add_executable(abser_ui_smoke tests/ui_smoke.cpp)
        target_compile_definitions(abser_ui_smoke PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
        target_link_libraries(abser_ui_smoke PRIVATE user32 shell32 gdi32 ole32 uuid)
        add_test(NAME ui COMMAND abser_ui_smoke $<TARGET_FILE:Abser>)
        set_tests_properties(ui PROPERTIES TIMEOUT 30)
        if(MSVC)
            target_compile_options(abser_ui_smoke PRIVATE /W4 /permissive- /utf-8)
        endif()
        add_executable(abser_file_drag_tests tests/file_drag_tests.cpp src/file_drag.cpp)
        target_include_directories(abser_file_drag_tests PRIVATE src)
        target_compile_definitions(abser_file_drag_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
        target_link_libraries(abser_file_drag_tests PRIVATE shell32 ole32 uuid)
        add_test(NAME file_drag COMMAND abser_file_drag_tests)
        set_tests_properties(file_drag PROPERTIES TIMEOUT 30)
        if(MSVC)
            target_compile_options(abser_file_drag_tests PRIVATE /W4 /permissive- /utf-8)
        endif()
    endif()
    if(MSVC)
        target_compile_options(Abser PRIVATE /W4 /permissive- /utf-8 /EHsc)
        target_link_options(Abser PRIVATE /MANIFEST:NO)
    elseif(MINGW)
        target_compile_options(Abser PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
        target_link_options(Abser PRIVATE -municode -Wl,--subsystem,windows -static -static-libgcc -static-libstdc++)
        if(BUILD_TESTING)
            target_link_options(catalog_tests PRIVATE -static -static-libgcc -static-libstdc++)
            target_link_options(abser_audio_tests PRIVATE -static -static-libgcc -static-libstdc++)
            target_link_options(abser_ui_smoke PRIVATE -municode -static -static-libgcc -static-libstdc++)
            target_link_options(abser_file_drag_tests PRIVATE -static -static-libgcc -static-libstdc++)
        endif()
    endif()
    install(TARGETS Abser RUNTIME DESTINATION .)
    install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
endif()
